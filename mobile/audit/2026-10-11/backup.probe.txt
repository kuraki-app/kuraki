import { beforeEach, expect, it, vi } from 'vitest';
const h = vi.hoisted(() => ({server:{baseURL:'https://a.test',deviceToken:'A'},done:new Set<string>(),uploads:[] as string[],networkReads:0,assets:[{id:'local-1',filename:'photo.jpg',uri:'file:///photo.jpg',creationTime:1577836800000}]}));
vi.mock('expo-media-library/legacy',()=>({requestPermissionsAsync:async()=>({granted:true}),getPermissionsAsync:async()=>({granted:true}),getAssetsAsync:async()=>({assets:h.assets,hasNextPage:false})}));
vi.mock('@/lib/capture-api',()=>({CaptureAPIError:class extends Error{},uploadFile:vi.fn(async(s)=>{h.uploads.push(s.baseURL);return{id:'upload',status:'queued',job_id:'not-imported-yet'};})}));
vi.mock('@/lib/backup-store',()=>({loadBackupState:async()=>({auto:true,failed:[],lastSuccess:null,albumIds:[],wifiOnly:true,legacyDoneIds:[]}),saveBackupState:async()=>{}}));
vi.mock('@/lib/backup-ledger',()=>({clearResumableUpload:async()=>{},importLegacyDoneIds:async()=>{},loadBackedUpIds:async()=>new Set(h.done),loadResumableUpload:async()=>null,markBackedUp:async(id:string)=>{h.done.add(id)},saveResumableUpload:async()=>{}}));
vi.mock('@/lib/settings',()=>({loadCaptureSettings:async()=>h.server}));
vi.mock('@/lib/network',()=>({currentConnection:async()=>{h.networkReads++;return{type:h.networkReads===1?'wifi':'cellular',isConnected:true}},evaluateNetworkGate:(n:any)=>n.type==='cellular'?'metered':'allowed',gateMessage:()=>null}));
vi.mock('@/lib/notifications',()=>({notify:async()=>{}}));
vi.mock('@/lib/prefs',()=>({loadPrefs:async()=>({}),mediaTypesFor:()=>['photo','video']}));
import {reclaimable} from '@/lib/reclaim';
beforeEach(()=>{vi.resetModules();h.server={baseURL:'https://a.test',deviceToken:'A'};h.done.clear();h.uploads=[];h.networkReads=0;h.assets=[{id:'local-1',filename:'photo.jpg',uri:'file:///photo.jpg',creationTime:1577836800000}];});
it('reproduces queued receipt becoming eligible for local removal before import',async()=>{const{backupEngine}=await import('@/lib/backup-engine');await backupEngine.run();expect(backupEngine.snapshot().done).toBe(1);expect(h.done.has('local-1')).toBe(true);expect(reclaimable(h.assets,h.done,7).length).toBe(1);console.log('REPRO: queued-only receipt marked backed-up and eligible for removal');});
it('reproduces a new server skipped by the shared backup ledger',async()=>{const first=await import('@/lib/backup-engine');await first.backupEngine.run();h.server={baseURL:'https://b.test',deviceToken:'B'};h.networkReads=0;vi.resetModules();const second=await import('@/lib/backup-engine');await second.backupEngine.run();expect(h.uploads).toEqual(['https://a.test']);expect(second.backupEngine.snapshot().done).toBe(1);console.log('REPRO: B says backed-up without receiving local photo');});
it('reproduces network checked once for a multiple-file run',async()=>{h.assets=[1,2,3].map(id=>({id:'local-'+id,filename:'photo.jpg',uri:'file:///photo.jpg',creationTime:1577836800000}));const{backupEngine}=await import('@/lib/backup-engine');await backupEngine.run();expect(h.uploads.length).toBe(3);expect(h.networkReads).toBe(1);console.log('REPRO: wifi-only policy never rechecked during 3 uploads');});

it('reproduces two overlapping singleton runs passing the preflight guard',async()=>{vi.doMock('@/lib/network',()=>({currentConnection:async()=>({type:'wifi',isConnected:true}),evaluateNetworkGate:()=> 'allowed',gateMessage:()=>null}));const{backupEngine}=await import('@/lib/backup-engine');await backupEngine.isAuto();await Promise.all([backupEngine.run(),backupEngine.run()]);expect(h.uploads.length).toBe(2);console.log('REPRO: two overlapping runs upload the same local photo');});
